```python
"""
Mở rộng: phân tích tên (Thần số học + Ngũ cách) và giờ sinh (Can Chi giờ).
"""
from datetime import datetime, date
from typing import Dict, List
import unicodedata


# ============ XỬ LÝ TÊN ============
# Bảng quy đổi chữ cái → số (Pythagoras)
PYTHAGORAS = {
    "A":1,"B":2,"C":3,"D":4,"E":5,"F":6,"G":7,"H":8,"I":9,
    "J":1,"K":2,"L":3,"M":4,"N":5,"O":6,"P":7,"Q":8,"R":9,
    "S":1,"T":2,"U":3,"V":4,"W":5,"X":6,"Y":7,"Z":8,
}
# Chữ cái nguyên âm
NGUYEN_AM = set("AEIOUY")

# Ngũ hành của chữ cái theo Hán Việt (bảng tra cứu phổ biến)
CHU_HANH = {
    # Mộc
    "A":"Mộc","Ă":"Mộc","Â":"Mộc","E":"Mộc","Ê":"Mộc","I":"Mộc","Y":"Mộc",
    # Hỏa
    "C":"Hỏa","D":"Hỏa","Đ":"Hỏa","G":"Hỏa","L":"Hỏa","N":"Hỏa","T":"Hỏa",
    "S":"Hỏa","X":"Hỏa",
    # Thổ
    "B":"Thổ","M":"Thổ","P":"Thổ","V":"Thổ",
    # Kim
    "K":"Kim","H":"Kim","Q":"Kim","R":"Kim",
    # Thủy
    "O":"Thủy","Ô":"Thủy","Ơ":"Thủy","U":"Thủy","Ư":"Thủy",
}


def chuan_hoa_ten(ten: str) -> str:
    """Bỏ dấu tiếng Việt để tra cứu số"""
    ten = ten.upper().strip()
    # Bỏ dấu nhưng giữ Đ
    normalized = unicodedata.normalize("NFD", ten)
    return "".join(c for c in normalized if unicodedata.category(c) != "Mn")


def tinh_so_linh_hon(ten: str) -> int:
    """
    Số linh hồn = tổng các nguyên âm.
    Đại diện cho khao khát nội tâm.
    """
    ten_norm = chuan_hoa_ten(ten)
    total = sum(PYTHAGORAS.get(c, 0) for c in ten_norm if c in NGUYEN_AM)
    while total > 9 and total not in (11, 22, 33):
        total = sum(int(d) for d in str(total))
    return total


def tinh_so_nhan_cach(ten: str) -> int:
    """
    Số nhân cách = tổng các phụ âm.
    Cách người khác nhìn bạn.
    """
    ten_norm = chuan_hoa_ten(ten)
    total = sum(PYTHAGORAS.get(c, 0)
                 for c in ten_norm if c.isalpha() and c not in NGUYEN_AM)
    while total > 9 and total not in (11, 22, 33):
        total = sum(int(d) for d in str(total))
    return total


def tinh_so_su_menh(ten: str) -> int:
    """
    Số sứ mệnh = tổng tất cả chữ cái (bao gồm cả âm và phụ âm).
    Mục đích cuộc đời.
    """
    ten_norm = chuan_hoa_ten(ten)
    total = sum(PYTHAGORAS.get(c, 0) for c in ten_norm if c.isalpha())
    while total > 9 and total not in (11, 22, 33):
        total = sum(int(d) for d in str(total))
    return total


def tinh_nam_cach(ho_ten: str) -> Dict:
    """
    Ngũ cách (theo Danh học Trung Hoa):
    - Thiên cách (天格): họ + 1
    - Nhân cách (人格): họ + tên đệm
    - Địa cách (地格): tên đệm + tên chính
    - Ngoại cách (外格): tổng - nhân cách + 1
    - Tổng cách (總格): tổng tất cả
    """
    parts = ho_ten.strip().split()
    if len(parts) < 2:
        parts = [ho_ten, ""]

    def so_chu(s: str) -> int:
        s = chuan_hoa_ten(s)
        return sum(PYTHAGORAS.get(c, 0) for c in s if c.isalpha())

    ho = parts[0]
    ten_dem = " ".join(parts[1:-1]) if len(parts) > 2 else ""
    ten_chinh = parts[-1]

    ho_so = so_chu(ho)
    dem_so = so_chu(ten_dem) if ten_dem else 0
    ten_so = so_chu(ten_chinh)
    tong = ho_so + dem_so + ten_so

    thien_cach = ho_so + 1
    nhan_cach = ho_so + dem_so if dem_so else ho_so + ten_so
    dia_cach = dem_so + ten_so if dem_so else ten_so + 1
    ngoai_cach = tong - nhan_cach + 1
    tong_cach = tong

    def to_1_chu_so(x: int) -> int:
        while x > 9 and x not in (11, 22, 33):
            x = sum(int(d) for d in str(x))
        return x

    return {
        "thien_cach": {"so": thien_cach, "rut_gon": to_1_chu_so(thien_cach),
                        "y_nghia": "Tổ tiên, di truyền, vận mệnh ban đầu"},
        "nhan_cach": {"so": nhan_cach, "rut_gon": to_1_chu_so(nhan_cach),
                       "y_nghia": "Bản thân, tính cách, vận mệnh chính"},
        "dia_cach": {"so": dia_cach, "rut_gon": to_1_chu_so(dia_cach),
                      "y_nghia": "Nền tảng, tuổi trẻ, vợ chồng"},
        "ngoai_cach": {"so": ngoai_cach, "rut_gon": to_1_chu_so(ngoai_cach),
                        "y_nghia": "Xã hội, bạn bè, quan hệ ngoài"},
        "tong_cach": {"so": tong_cach, "rut_gon": to_1_chu_so(tong_cach),
                       "y_nghia": "Tổng vận mệnh, hậu vận"},
    }


def ngu_hanh_ten(ten: str) -> Dict:
    """Phân tích ngũ hành của tên"""
    ten_norm = chuan_hoa_ten(ten)
    dem = {"Kim": 0, "Mộc": 0, "Thủy": 0, "Hỏa": 0, "Thổ": 0}
    for c in ten_norm:
        hanh = CHU_HANH.get(c)
        if hanh:
            dem[hanh] += 1
    tong = sum(dem.values()) or 1
    ty_le = {k: round(v * 100 / tong, 1) for k, v in dem.items()}
    hanh_chinh = max(dem, key=dem.get)
    return {
        "phan_bo": dem,
        "ty_le_phan_tram": ty_le,
        "hanh_chinh": hanh_chinh,
        "thieu": [k for k, v in dem.items() if v == 0],
    }


def phan_tich_ten(ho_ten: str, menh_nap_am: str = None) -> Dict:
    """Phân tích tên đầy đủ"""
    if not ho_ten.strip():
        return {"error": "Chưa có tên"}

    linh_hon = tinh_so_linh_hon(ho_ten)
    nhan_cach = tinh_so_nhan_cach(ho_ten)
    su_menh = tinh_so_su_menh(ho_ten)
    nam_cach = tinh_nam_cach(ho_ten)
    ngu_hanh = ngu_hanh_ten(ho_ten)

    # Đánh giá hợp mệnh
    hop_menh = None
    if menh_nap_am:
        from lich_van_nien import quan_he_ngu_hanh
        qh = quan_he_ngu_hanh(menh_nap_am, ngu_hanh["hanh_chinh"])
        hop_menh = {
            "menh_chu": menh_nap_am,
            "hanh_ten": ngu_hanh["hanh_chinh"],
            "quan_he": qh,
            "diem": {"tương sinh": 90, "tỷ hòa": 75,
                      "trung tính": 60, "tương khắc": 35}[qh],
            "nhan_xet": f"Tên hợp mệnh {menh_nap_am}" if qh in ("tương sinh", "tỷ hòa")
                        else f"Tên khắc mệnh {menh_nap_am} — cân nhắc đặt tên thêm"
                             if qh == "tương khắc" else "Tên trung tính với mệnh",
        }

    return {
        "ho_ten": ho_ten,
        "so_linh_hon": linh_hon,
        "so_nhan_cach": nhan_cach,
        "so_su_menh": su_menh,
        "nam_cach": nam_cach,
        "ngu_hanh_ten": ngu_hanh,
        "hop_menh": hop_menh,
    }


# ============ XỬ LÝ GIỜ SINH ============
# 12 Canh giờ
CANH_GIO = [
    ("Tý", 23, 1), ("Sửu", 1, 3), ("Dần", 3, 5), ("Mão", 5, 7),
    ("Thìn", 7, 9), ("Tỵ", 9, 11), ("Ngọ", 11, 13), ("Mùi", 13, 15),
    ("Thân", 15, 17), ("Dậu", 17, 19), ("Tuất", 19, 21), ("Hợi", 21, 23),
]


def xac_dinh_canh_gio(gio: int) -> Dict:
    """Xác định Canh giờ từ giờ (0-23)"""
    if gio == 23 or gio == 0:
        chi_idx = 0  # Tý
    else:
        chi_idx = ((gio + 1) // 2) % 12

    ten, start, end = CANH_GIO[chi_idx]
    return {
        "gio": gio,
        "canh": ten,
        "chi_idx": chi_idx,
        "khoang_thoi_gian": f"{start}h - {end}h",
    }


def tinh_can_gio(can_ngay: str, gio: int) -> str:
    """
    Tính Can của giờ dựa trên Can của ngày.
    Quy tắc: Ngày Giáp/Kỷ → giờ Tý là Giáp
             Ngày Ất/Canh → giờ Tý là Bính
             Ngày Bính/Tân → giờ Tý là Mậu
             Ngày Đinh/Nhâm → giờ Tý là Canh
             Ngày Mậu/Quý → giờ Tý là Nhâm
    """
    from lich_van_nien import CAN
    # Chỉ số Can ngày
    can_ngay_idx = CAN.index(can_ngay)

    # Xác định Can giờ Tý
    can_ty_idx = {
        0: 0, 5: 0,  # Giáp, Kỷ → Giáp
        1: 2, 6: 2,  # Ất, Canh → Bính
        2: 4, 7: 4,  # Bính, Tân → Mậu
        3: 6, 8: 6,  # Đinh, Nhâm → Canh
        4: 8, 9: 8,  # Mậu, Quý → Nhâm
    }[can_ngay_idx]

    # Canh giờ (0=Tý, 1=Sửu, ...)
    canh = xac_dinh_canh_gio(gio)
    can_gio_idx = (can_ty_idx + canh["chi_idx"]) % 10

    return f"{CAN[can_gio_idx]} {canh['canh']}"


def phan_tich_gio_sinh(ngay: int, thang: int, nam: int, gio: int) -> Dict:
    """Phân tích giờ sinh đầy đủ"""
    from lich_van_nien import ngay_can_chi, CAN_NGU_HANH, CHI_NGU_HANH

    d = date(nam, thang, ngay)
    can_ngay, chi_ngay = ngay_can_chi(d)

    can_gio_chi = tinh_can_gio(can_ngay, gio)

        "Mão": "Dịu dàng, khéo léo, yêu cái đẹp, hay lo xa",
        "Thìn": "Uy nghi, tham vọng, có tố chất lãnh đạo",
        "Tỵ": "Sâu sắc, kín đáo, trực giác mạnh, khéo nói",
        "Ngọ": "Nhiệt huyết, thích tự do, hay đi xa, bốc đồng",
        "Mùi": "Hiền lành, nhạy cảm, có khiếu nghệ thuật",
        "Thân": "Linh hoạt, đa tài, thích khám phá, hay thay đổi",
        "Dậu": "Tỉ mỉ, cầu toàn, thẳng thắn, trọng chữ tín",
```

```python
def phan_tich_ngay_sinh(day: int, month: int, year: int,
                          ho_ten: str = "", gio_sinh: int = None) -> Dict:
    """Phân tích toàn diện: ngày + tên + giờ sinh"""
    can_chi = tinh_can_chi_nam(year)
    can = can_chi.split()[0]
    chi = can_chi.split()[1]

    profile = {
        "ho_ten": ho_ten,
        "ngay_sinh": f"{day:02d}/{month:02d}/{year}",
        "can_chi_nam": can_chi,
        "ngu_hanh_can": CAN_NGU_HANH[can],
        "ngu_hanh_chi": CHI_NGU_HANH[chi],
        "menh_nap_am": tinh_menh_ngu_hanh(year),
        "so_chu_dao": tinh_so_chu_dao(day, month, year),
        "cung_phi": tinh_cung_phi(day, month, year),
        "bat_quai": BAT_QUAI[tinh_cung_phi(day, month, year)],
        "tam_tai": tinh_tam_tai(day, month, year),
        "con_giap": chi,
    }

    # Thêm tên
    if ho_ten:
        profile["ten_phan_tich"] = phan_tich_ten(ho_ten, profile["menh_nap_am"])

    # Thêm giờ sinh
    if gio_sinh is not None:
        profile["gio_sinh_phan_tich"] = phan_tich_gio_sinh(
            day, month, year, gio_sinh
        )

    return profile
```

### `deploy/cloudflare/worker.js`

```javascript
/**
 * Cloudflare Worker cho huyenbao.com
 * - Proxy API về backend
 * - Serve static assets từ KV
 * - Rate limiting
 * - Cache layer
 * - CORS
 */

const BACKEND_ORIGIN = "https://api.huyenbao.com";
const STATIC_DOMAIN = "static.huyenbao.com";

addEventListener("fetch", event => {
  event.respondWith(handleRequest(event));
});

async function handleRequest(event) {
  const request = event.request;
  const url = new URL(request.url);
  const host = url.hostname;

  // 1. CORS preflight
  if (request.method === "OPTIONS") {
    return corsResponse();
  }

  // 2. Rate limiting theo IP
  const ip = request.headers.get("CF-Connecting-IP") || "unknown";
  const rateLimit = await checkRateLimit(ip, url.pathname);
  if (rateLimit.exceeded) {
    return jsonResponse({
      error: "Quá nhiều yêu cầu. Vui lòng thử lại sau.",
      retry_after: rateLimit.retry_after,
    }, 429);
  }

  // 3. Route theo path
  if (url.pathname.startsWith("/api/")) {
    return proxyToBackend(request, url);
  }

  if (url.pathname.startsWith("/static/") ||
      url.pathname.match(/\.(js|css|png|jpg|svg|woff2|json)$/)) {
    return serveStatic(request, url);
  }

  // 4. Mặc định: serve index.html (SPA routing)
  return serveSPA(request);
}

// ============ RATE LIMITING ============
async function checkRateLimit(ip, path) {
  // Giới hạn khác nhau theo path
  let limit = 100, window = 60; // mặc định: 100 req/phút
  if (path.startsWith("/api/auth/")) {
    limit = 5; window = 60; // 5 req/phút cho auth
  } else if (path.match(/\/api\/(analyze|tarot|kinh-dich|tu-vi|video)/)) {
    limit = 10; window = 60; // 10 req/phút cho LLM endpoints
  }

  const key = `ratelimit:${ip}:${Math.floor(Date.now() / (window * 1000))}`;
  const count = await RATE_LIMIT.get(key);
  const current = count ? parseInt(count) : 0;

  if (current >= limit) {
    return { exceeded: true, retry_after: window };
  }

  await RATE_LIMIT.put(key, String(current + 1), {
    expirationTtl: window * 2,
  });
  return { exceeded: false };
}

// ============ PROXY ============
async function proxyToBackend(request, url) {
  const targetUrl = BACKEND_ORIGIN + url.pathname + url.search;

  // Copy headers, thêm CF headers
  const headers = new Headers(request.headers);
  headers.set("X-Forwarded-For", request.headers.get("CF-Connecting-IP"));
  headers.set("X-Real-IP", request.headers.get("CF-Connecting-IP"));
  headers.set("X-CF-Worker", "huyenbao-v1");

  // Với SSE stream, cần disable cache + timeout dài
  const isStream = url.pathname.includes("/stream");
  const fetchOptions = {
    method: request.method,
    headers,
    body: request.body,
    redirect: "follow",
  };
  if (isStream) {
    fetchOptions.cf = {
      cacheTtl: 0,
      cacheEverything: false,
      scrapeShield: false,
    };
  }

  const response = await fetch(targetUrl, fetchOptions);

  // Thêm CORS headers
  const newHeaders = new Headers(response.headers);
  newHeaders.set("Access-Control-Allow-Origin", "*");
  newHeaders.set("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS");
  newHeaders.set("Access-Control-Allow-Headers", "*");
  newHeaders.set("X-Worker", "huyenbao");

  return new Response(response.body, {
    status: response.status,
    statusText: response.statusText,
    headers: newHeaders,
  });
}

// ============ STATIC ============
async function serveStatic(request, url) {
  // Cache-first từ KV
  const cacheKey = url.pathname;
  const cached = await STATIC_KV.get(cacheKey, { type: "arrayBuffer" });

  if (cached) {
    return new Response(cached, {
      headers: {
        "Content-Type": getContentType(url.pathname),
        "Cache-Control": "public, max-age=31536000, immutable",
        "X-Cache": "HIT",
      },
    });
  }

  // Fallback: fetch từ origin
  const originUrl = `https://${STATIC_DOMAIN}${url.pathname}`;
  const response = await fetch(originUrl);
  if (response.ok) {
    const buffer = await response.arrayBuffer();
    await STATIC_KV.put(cacheKey, buffer, {
      expirationTtl: 86400 * 30,
    });
    return new Response(buffer, {
      headers: {
        "Content-Type": getContentType(url.pathname),
        "Cache-Control": "public, max-age=31536000, immutable",
        "X-Cache": "MISS",
      },
    });
  }
  return new Response("Not Found", { status: 404 });
}

async function serveSPA(request) {
  // Serve index.html từ KV hoặc origin
  const cached = await STATIC_KV.get("/index.html", { type: "arrayBuffer" });
  if (cached) {
    return new Response(cached, {
      headers: {
        "Content-Type": "text/html; charset=utf-8",
        "Cache-Control": "public, max-age=300",
      },
    });
  }
  const response = await fetch(`https://${STATIC_DOMAIN}/index.html`);
  return response;
}

// ============ HELPERS ============
function corsResponse() {
  return new Response(null, {
    status: 204,
    headers: {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, POST, PUT, DELETE, OPTIONS",
      "Access-Control-Allow-Headers": "*",
      "Access-Control-Max-Age": "86400",
    },
  });
}

function jsonResponse(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      "Content-Type": "application/json",
      "Access-Control-Allow-Origin": "*",
    },
  });
}

function getContentType(path) {
  const ext = path.split(".").pop().toLowerCase();
  const types = {
    html: "text/html; charset=utf-8",
    js: "application/javascript",
    css: "text/css",
    json: "application/json",
    png: "image/png",
    jpg: "image/jpeg",
    jpeg: "image/jpeg",
    svg: "image/svg+xml",
    webp: "image/webp",
    woff: "font/woff",
    woff2: "font/woff2",
    mp4: "video/mp4",
    pdf: "application/pdf",
    txt: "text/plain; charset=utf-8",
  };
  return types[ext] || "application/octet-stream";
}
```

```bash
#!/bin/bash
# Deploy Cloudflare Worker cho huyenbao.com

set -e

echo "🌐 Deploying Cloudflare Worker..."

# 1. Cài wrangler
if ! command -v wrangler &> /dev/null; then
    echo "📦 Installing wrangler..."
```

```python
def tao_tenant(db: Session, ten_doanh_nghiep: str, email: str,
                 slug: str, tier: str = "basic") -> Dict:
    """Tạo tenant mới"""
    if db.query(Tenant).filter_by(slug=slug).first():
        raise ValueError(f"Slug '{slug}' đã tồn tại")

    tier_info = TIERS[tier]
    tenant = Tenant(
        slug=slug, ten_doanh_nghiep=ten_doanh_nghiep,
        email_lien_he=email, tier=tier,
        max_users=tier_info["max_users"],
        max_llm_calls_month=tier_info["max_llm_calls_month"],
        features_json=__import__("json").dumps(tier_info["features"]),
    )
    db.add(tenant); db.commit(); db.refresh(tenant)

    # Tạo API key đầu tiên
    api_key = tao_api_key(db, tenant.id, "production")
    return {"tenant_id": tenant.id, "slug": tenant.slug,
            "api_key": api_key["key"], "tier": tier}


def tao_api_key(db: Session, tenant_id: int, name: str) -> Dict:
    """Tạo API key cho tenant"""
    import hashlib
    key = "hb_" + secrets.token_urlsafe(32)
    key_hash = hashlib.sha256(key.encode()).hexdigest()

    api_key = ApiKey(tenant_id=tenant_id, key_hash=key_hash, name=name)
    db.add(api_key); db.commit(); db.refresh(api_key)

    # Chỉ trả key 1 lần — sau đó không thể lấy lại
    return {"id": api_key.id, "key": key, "name": name,
            "luu_y": "Lưu key này ngay, không thể xem lại!"}


def xac_thuc_api_key(db: Session, key: str) -> Dict:
    """Xác thực API key và trả tenant"""
    import hashlib
    key_hash = hashlib.sha256(key.encode()).hexdigest()
    api_key = db.query(ApiKey).filter_by(key_hash=key_hash, is_active=True).first()
    if not api_key:
        return {"valid": False, "reason": "API key không hợp lệ"}

    tenant = db.query(Tenant).filter_by(id=api_key.tenant_id, is_active=True).first()
    if not tenant:
        return {"valid": False, "reason": "Tenant không hoạt động"}

    # Update usage
    api_key.last_used = datetime.utcnow()
    api_key.calls_count += 1
    db.commit()

    # Check quota
    thang = datetime.now().strftime("%Y-%m")
    usage = db.query(TenantUsage).filter_by(
        tenant_id=tenant.id, thang=thang
    ).first()
    if not usage:
        usage = TenantUsage(tenant_id=tenant.id, thang=thang)
        db.add(usage); db.commit()

    if usage.llm_calls >= tenant.max_llm_calls_month:
        return {"valid": False, "reason": "Đã vượt quota LLM tháng này",
                "tenant": tenant, "quota_exceeded": True}

    features = __import__("json").loads(tenant.features_json or "{}")
    return {
        "valid": True,
        "tenant": {
            "id": tenant.id, "slug": tenant.slug,
            "ten_doanh_nghiep": tenant.ten_doanh_nghiep,
            "brand_name": tenant.brand_name,
            "logo_url": tenant.logo_url,
            "primary_color": tenant.primary_color,
            "secondary_color": tenant.secondary_color,
            "footer_text": tenant.footer_text,
            "tier": tenant.tier,
            "features": features,
        },
    }


def ghi_usage(db: Session, tenant_id: int, loai: str, so_luong: int = 1):
    """Ghi nhận sử dụng"""
    thang = datetime.now().strftime("%Y-%m")
    usage = db.query(TenantUsage).filter_by(
        tenant_id=tenant_id, thang=thang
    ).first()
    if not usage:
        usage = TenantUsage(tenant_id=tenant_id, thang=thang)
        db.add(usage)
    if loai == "llm":
        usage.llm_calls += so_luong
    elif loai == "video":
        usage.video_renders += so_luong
    elif loai == "pdf":
        usage.pdf_exports += so_luong
    db.commit()


def bao_cao_tenant(db: Session, tenant_id: int) -> Dict:
    """Báo cáo sử dụng cho tenant"""
    tenant = db.query(Tenant).filter_by(id=tenant_id).first()
    if not tenant:
        return {"error": "Không tìm thấy tenant"}

    thang_nay = datetime.now().strftime("%Y-%m")

    ).first()
```

```python
# backend/middleware_tenant.py
from fastapi import Request, HTTPException, Depends
from fastapi.security import APIKeyHeader
from sqlalchemy.orm import Session
from database import SessionLocal
from white_label import xac_thuc_api_key

api_key_header = APIKeyHeader(name="X-API-Key", auto_error=False)


def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


async def get_tenant_from_api_key(
    api_key: str = Depends(api_key_header),
    db: Session = Depends(get_db),
):
    """Middleware xác thực API key cho white-label"""
    if not api_key:
        raise HTTPException(401, "Thiếu X-API-Key header")
    result = xac_thuc_api_key(db, api_key)
    if not result["valid"]:
        raise HTTPException(401, result.get("reason", "API key không hợp lệ"))
    return result["tenant"]
```

```python
# ============ PLATFORM UPLOADERS ============
class TikTokUploader:
    """TikTok Content Posting API"""

    def __init__(self, access_token: str):
        self.token = access_token
        self.base = "https://open.tiktokapis.com/v2"

    async def upload_video(self, video_path: str, caption: str) -> Dict:
        """Upload video lên TikTok"""
        async with httpx.AsyncClient(timeout=300) as client:
            # 1. Init upload
            init_r = await client.post(
                f"{self.base}/post/publish/video/init/",
                headers={"Authorization": f"Bearer {self.token}"},
                json={
                    "post_info": {
                        "title": caption[:2200],
                        "privacy_level": "PUBLIC_TO_EVERYONE",
                        "disable_duet": False,
                        "disable_comment": False,
                        "disable_stitch": False,
                    },
                    "source_info": {
                        "source": "FILE_UPLOAD",
                        "video_size": os.path.getsize(video_path),
                        "chunk_size": os.path.getsize(video_path),
                        "total_chunk_count": 1,
                    },
                },
            )
            data = init_r.json()
            if "data" not in data or "upload_url" not in data.get("data", {}):
                return {"error": "Không tạo được upload URL", "detail": data}

            upload_url = data["data"]["upload_url"]
            publish_id = data["data"]["publish_id"]

            # 2. Upload file
            with open(video_path, "rb") as f:
                video_bytes = f.read()

            upload_r = await client.put(
                upload_url,
                content=video_bytes,
                headers={
                    "Content-Type": "video/mp4",
                    "Content-Range": f"bytes 0-{len(video_bytes)-1}/{len(video_bytes)}",
                },
            )
            if upload_r.status_code not in (200, 201, 206):
                return {"error": "Upload thất bại", "status": upload_r.status_code}

            return {"success": True, "publish_id": publish_id}


class YouTubeUploader:
    """YouTube Data API v3"""

    def __init__(self, access_token: str):
        self.token = access_token
        self.base = "https://www.googleapis.com/upload/youtube/v3"

    async def upload_video(self, video_path: str, title: str,
                            description: str, tags: List[str] = None,
                            privacy: str = "public") -> Dict:
        """Upload video lên YouTube (resumable)"""
        metadata = {
            "snippet": {
                "title": title[:100],
                "description": description[:5000],
                "tags": tags or [],
                "categoryId": "22",  # People & Blogs
            },
            "status": {
                "privacyStatus": privacy,
                "selfDeclaredMadeForKids": False,
            },
        }

        async with httpx.AsyncClient(timeout=600) as client:
            # 1. Init resumable upload
            init_r = await client.post(
                f"{self.base}/videos?uploadType=resumable&part=snippet,status",
                headers={
                    "Authorization": f"Bearer {self.token}",
                    "Content-Type": "application/json",
                    "X-Upload-Content-Type": "video/mp4",
                    "X-Upload-Content-Length": str(os.path.getsize(video_path)),
                },
                json=metadata,
            )
            if init_r.status_code != 200:
                return {"error": "Không tạo được session", "detail": init_r.text}

            upload_url = init_r.headers.get("Location")

            # 2. Upload video
            with open(video_path, "rb") as f:
                video_bytes = f.read()

            upload_r = await client.put(
                upload_url,
                content=video_bytes,
                headers={
                    "Content-Type": "video/mp4",
                    "Content-Length": str(len(video_bytes)),
                },
            )
            if upload_r.status_code not in (200, 201):
                return {"error": "Upload thất bại", "detail": upload_r.text}

            data = upload_r.json()
            return {
                "success": True,
                "video_id": data.get("id"),
                "url": f"https://youtube.com/watch?v={data.get('id')}",
            }


class InstagramUploader:
    """Instagram Graph API (Business accounts)"""

    def __init__(self, access_token: str, ig_user_id: str):
        self.token = access_token
        self.ig_user_id = ig_user_id
        self.base = "https://graph.facebook.com/v18.0"

    async def upload_reel(self, video_url: str, caption: str) -> Dict:
        """Upload Reel — cần public URL của video trước"""
        async with httpx.AsyncClient(timeout=300) as client:
            # 1. Create container
            create_r = await client.post(
                f"{self.base}/{self.ig_user_id}/media",
                params={
                    "media_type": "REELS",
                    "video_url": video_url,
                    "caption": caption,
                    "access_token": self.token,
                },
            )
            container_id = create_r.json().get("id")
            if not container_id:
                return {"error": "Không tạo được container", "detail": create_r.json()}

            # 2. Publish (đợi vài giây cho container xử lý)
            import asyncio
            await asyncio.sleep(5)
            pub_r = await client.post(
                f"{self.base}/{self.ig_user_id}/media_publish",
                params={"creation_id": container_id, "access_token": self.token},
            )
            return {"success": True, "post_id": pub_r.json().get("id")}


# ============ ORCHESTRATOR ============
async def post_to_multiple_platforms(
    db: Session, user_id: int, video_path: str,
    caption: str, platforms: List[str],
) -> Dict:
    """
    Đăng video lên nhiều nền tảng cùng lúc.
    """
    accounts = db.query(SocialAccount).filter(
        SocialAccount.user_id == user_id,
        SocialAccount.platform.in_(platforms),
        SocialAccount.is_active == True,
    ).all()

    if not accounts:
        return {"error": "Chưa có tài khoản social nào được kết nối"}

    results = {}
    for account in accounts:
        try:
            if account.platform == "tiktok":
                uploader = TikTokUploader(account.access_token)
                results["tiktok"] = await uploader.upload_video(video_path, caption)
            elif account.platform == "youtube":
                uploader = YouTubeUploader(account.access_token)
                results["youtube"] = await uploader.upload_video(
                    video_path,
                    title=caption.split("\n")[0][:100],
                    description=caption,
                )
            elif account.platform == "instagram":
                # Instagram cần public URL
                # Upload video lên CDN trước (huyenbao.com)
                video_url = f"https://huyenbao.com/media/{os.path.basename(video_path)}"
                uploader = InstagramUploader(account.access_token,
                                               account.account_name)
                results["instagram"] = await uploader.upload_reel(video_url, caption)
        except Exception as e:
            results[account.platform] = {"error": str(e)}

    return results


def lap_lich_dang(db: Session, user_id: int, video_path: str,
                    caption: str, platforms: List[str],
                    scheduled_at: datetime = None) -> Dict:
    """Lập lịch đăng video"""
    post = ScheduledPost(
        user_id=user_id, video_path=video_path, caption=caption,
        hashtags="", platforms_json=json.dumps(platforms),
        scheduled_at=scheduled_at or datetime.utcnow(),
    )
    db.add(post); db.commit(); db.refresh(post)
    return {"post_id": post.id, "scheduled_at": post.scheduled_at.isoformat()}
```

```python
def lay_plan(user_id: int, db: Session) -> Dict:
    """Lấy plan hiện tại của user"""
    sub = db.query(Subscription).filter_by(
        user_id=user_id, status="active"
    ).first()

    if not sub or (sub.current_period_end and sub.current_period_end < datetime.utcnow()):
        return {"plan": "free", **PLANS["free"]}

    return {
        "plan": sub.plan,
        "current_period_end": sub.current_period_end.isoformat() if sub.current_period_end else None,
        "cancel_at_period_end": sub.cancel_at_period_end,
        **PLANS.get(sub.plan, PLANS["free"]),
    }


def check_quota(user_id: int, feature: str, db: Session) -> Dict:
    """Kiểm tra user còn quota không"""
    plan = lay_plan(user_id, db)
    limits = plan.get("limits", {})

    if feature == "llm_calls":
        limit = limits.get("llm_calls_day", 5)
        used = dem_su_dung_hom_nay(user_id, "llm", db)
        return {
            "allowed": used < limit,
            "used": used, "limit": limit,
            "remaining": max(0, limit - used),
        }

    if feature == "video_renders":
        limit = limits.get("video_renders_day", 0)
        used = dem_su_dung_hom_nay(user_id, "video", db)
        return {
            "allowed": used < limit,
            "used": used, "limit": limit,
            "remaining": max(0, limit - used),
        }

    return {"allowed": True}


def dem_su_dung_hom_nay(user_id: int, loai: str, db: Session) -> int:
    """Đếm số lần dùng hôm nay (dùng Prediction table làm nguồn)"""
    from database import Prediction
    today = datetime.utcnow().replace(hour=0, minute=0, second=0)
    if loai == "llm":
        return db.query(Prediction).filter(
            Prediction.created_at >= today
        ).count()
    return 0


async def tao_checkout_session(user: User, plan: str, db: Session) -> Dict:
    """Tạo Stripe Checkout Session"""
    if plan not in PLANS or plan == "free":
        raise ValueError(f"Plan không hợp lệ: {plan}")

    plan_info = PLANS[plan]

    # Tìm/tạo Stripe customer
    sub = db.query(Subscription).filter_by(user_id=user.id).first()
    if not sub or not sub.stripe_customer_id:
        customer = stripe.Customer.create(
            email=user.email,
            metadata={"user_id": user.id, "username": user.username},
        )
        if not sub:
            sub = Subscription(user_id=user.id,
                                stripe_customer_id=customer.id,
                                plan="free", status="active")
            db.add(sub)
        else:
            sub.stripe_customer_id = customer.id
        db.commit()

    # Tạo checkout session
    session = stripe.checkout.Session.create(
        customer=sub.stripe_customer_id,
        mode="subscription",
        payment_method_types=["card"],
        line_items=[{
            "price": plan_info["stripe_price_id"],
            "quantity": 1,
        }],
        success_url="https://huyenbao.com/payment/success?session_id={CHECKOUT_SESSION_ID}",
        cancel_url="https://huyenbao.com/payment/cancel",
        metadata={"user_id": user.id, "plan": plan},
    )
    return {"checkout_url": session.url, "session_id": session.id}


async def xu_ly_webhook(payload: bytes, sig_header: str, db: Session) -> Dict:
    """Xử lý webhook từ Stripe"""
    try:
        event = stripe.Webhook.construct_event(
            payload, sig_header, STRIPE_WEBHOOK_SECRET
        )
    except Exception as e:
        return {"error": str(e)}

    event_type = event["type"]
    data = event["data"]["object"]

    if event_type == "checkout.session.completed":
        user_id = int(data["metadata"]["user_id"])
        plan = data["metadata"]["plan"]
        sub_id = data["subscription"]

        # Lấy subscription từ Stripe
        stripe_sub = stripe.Subscription.retrieve(sub_id)

        sub = db.query(Subscription).filter_by(user_id=user_id).first()
        if sub:
            sub.stripe_subscription_id = sub_id
            sub.plan = plan
            sub.status = "active"
            sub.current_period_end = datetime.fromtimestamp(
                stripe_sub["current_period_end"]
            )
            db.commit()

        return {"handled": True, "action": "subscription_activated"}

    if event_type == "invoice.payment_succeeded":
        # Ghi payment
        customer_id = data.get("customer")
        sub = db.query(Subscription).filter_by(
            stripe_customer_id=customer_id
        ).first()
        if sub:
            db.add(Payment(
                user_id=sub.user_id,
                stripe_payment_id=data["id"],
                amount=data["amount_paid"] / 100,
                currency=data["currency"],
                status="succeeded",
                description=data.get("description"),
            ))
            db.commit()
        return {"handled": True, "action": "payment_recorded"}

    if event_type == "customer.subscription.deleted":
        sub = db.query(Subscription).filter_by(
            stripe_subscription_id=data["id"]
        ).first()
        if sub:
            sub.status = "canceled"
            sub.plan = "free"
            db.commit()
        return {"handled": True, "action": "subscription_canceled"}

    return {"handled": False, "event": event_type}
```

```python
# backend/middleware_quota.py
from fastapi import Depends, HTTPException
from sqlalchemy.orm import Session
from auth import get_current_user, get_db
from database import User
from monetization import check_quota


def require_quota(feature: str):
    """Dependency factory — kiểm tra quota cho feature"""
    def check(user: User = Depends(get_current_user),
              db: Session = Depends(get_db)):
        quota = check_quota(user.id, feature, db)
        if not quota["allowed"]:
            raise HTTPException(
                429,
                f"Hết quota {feature}. Còn {quota['remaining']}/{quota['limit']}. "
                f"Nâng cấp Premium để dùng không giới hạn."
            )
        return quota
    return check
```

```python
def detect_lang(request: Request) -> str:
    """Detect ngôn ngữ từ Accept-Language header hoặc ?lang="""
    # Query param ưu tiên
    lang = request.query_params.get("lang")
    if lang in LANG_SUPPORTED:
        return lang

    # Header Accept-Language
    accept = request.headers.get("accept-language", "")
    for part in accept.split(","):
        code = part.split(";")[0].strip()[:2].lower()
        if code in LANG_SUPPORTED:
            return code

    return LANG_DEFAULT


def t(key: str, lang: str = "vi", **kwargs) -> str:
    """Translate function"""
    lang_dict = TRANSLATIONS.get(lang, TRANSLATIONS[LANG_DEFAULT])
    text = lang_dict.get(key) or TRANSLATIONS[LANG_DEFAULT].get(key) or key
    if kwargs:
        try:
            text = text.format(**kwargs)
        except (KeyError, IndexError):
            pass
    return text


async def get_lang(request: Request) -> str:
    """Dependency: get ngôn ngữ từ request"""
    return detect_lang(request)
```

```python
def tao_metadata_nft(profile: Dict, chi_so: Dict,

# ============ MINT (dùng web3.py) ============
def mint_nft(user_id: int, profile_id: int, wallet_address: str,
               metadata: Dict, db: Session) -> Dict:
    """
    Mint NFT trên blockchain.
    Yêu cầu: WEB3_PROVIDER, CONTRACT_ADDRESS, PRIVATE_KEY.
    """
    try:
        from web3 import Web3
        from eth_account import Account
        from web3.middleware import geth_poa_middleware
    except ImportError:
        return {"error": "Cần cài: pip install web3 eth-account"}

    if not CONTRACT_ADDRESS or not PRIVATE_KEY:
        return {"error": "Chưa cấu hình contract hoặc private key"}

    w3 = Web3(Web3.HTTPProvider(WEB3_PROVIDER))
    if not w3.is_connected():
        return {"error": "Không kết nối được blockchain"}

    # Thêm middleware cho Polygon
    w3.middleware_onion.inject(geth_poa_middleware, layer=0)

    # ABI rút gọn của ERC-721
    ABI = json.loads("""
    [
      {"inputs":[{"name":"to","type":"address"},{"name":"uri","type":"string"}],
       "name":"safeMint","outputs":[{"name":"","type":"uint256"}],
       "stateMutability":"nonpayable","type":"function"}
    ]
    """)

    account = Account.from_key(PRIVATE_KEY)
    contract = w3.eth.contract(
        address=Web3.to_checksum_address(CONTRACT_ADDRESS),
        abi=ABI,
    )

    # Upload metadata lên IPFS (dùng Pinata/NFT.Storage)
    metadata_uri = upload_to_ipfs(metadata)
    if not metadata_uri:
        return {"error": "Không upload được metadata"}

    # Build tx
    nonce = w3.eth.get_transaction_count(account.address)
    tx = contract.functions.safeMint(
        Web3.to_checksum_address(wallet_address),
        metadata_uri,
    ).build_transaction({
        "from": account.address,
        "nonce": nonce,
        "gas": 300000,
        "maxFeePerGas": w3.to_wei("50", "gwei"),
        "maxPriorityFeePerGas": w3.to_wei("30", "gwei"),
        "chainId": w3.eth.chain_id,
    })

    signed = account.sign_transaction(tx)
    tx_hash = w3.eth.send_raw_transaction(signed.rawTransaction)
    receipt = w3.eth.wait_for_transaction_receipt(tx_hash, timeout=180)

    # Parse token_id từ event Transfer
    token_id = None
    for log in receipt.logs:
        if len(log["topics"]) == 4:  # Transfer event
            token_id = str(int(log["topics"][3].hex(), 16))
            break

    nft = LaSoNFT(
        user_id=user_id, profile_id=profile_id,
        token_id=token_id or "0",
        tx_hash=tx_hash.hex(),
        chain="polygon",
        metadata_uri=metadata_uri,
        metadata_json=json.dumps(metadata, ensure_ascii=False),
    )
    db.add(nft); db.commit(); db.refresh(nft)

    return {
        "token_id": nft.token_id,
        "tx_hash": nft.tx_hash,
        "chain": "polygon",
        "opensea_url": f"https://opensea.io/assets/matic/{CONTRACT_ADDRESS}/{nft.token_id}",
        "polygonscan_url": f"https://polygonscan.com/tx/{nft.tx_hash}",
    }


def upload_to_ipfs(metadata: Dict) -> str:
    """Upload metadata lên IPFS qua Pinata"""
    pinata_jwt = os.getenv("PINATA_JWT", "")
    if not pinata_jwt:
        # Fallback: lưu metadata và dùng URL huyenbao.com
        content_hash = hashlib.sha256(
            json.dumps(metadata, sort_keys=True).encode()
        ).hexdigest()[:32]
        # Lưu vào DB riêng hoặc file static
        return f"https://huyenbao.com/metadata/{content_hash}.json"

    import httpx
    r = httpx.post(
        "https://api.pinata.cloud/pinning/pinJSONToIPFS",
        headers={"Authorization": f"Bearer {pinata_jwt}"},
        json={"pinataContent": metadata},
        timeout=30,
    )
    if r.status_code == 200:
        ipfs_hash = r.json()["IpfsHash"]
        return f"ipfs://{ipfs_hash}"
    return ""
```

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC721/extensions/ERC721URIStorage.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract LaSoNFT is ERC721URIStorage, Ownable {
    uint256 private _nextTokenId;

    event LaSoMinted(address indexed to, uint256 indexed tokenId, string uri);

    constructor() ERC721("Huyen Bao La So", "HBLS") Ownable(msg.sender) {}

    function safeMint(address to, string memory uri)
        public
        onlyOwner
        returns (uint256)
    {
        uint256 tokenId = _nextTokenId++;
        _safeMint(to, tokenId);
        _setTokenURI(tokenId, uri);
        emit LaSoMinted(to, tokenId, uri);
        return tokenId;
    }
}
```

```python
Hỗ trợ ElevenLabs, Coqui TTS, hoặc Google Cloud TTS.

import os

# ============ ELEVENLABS ============
ELEVENLABS_VOICES = {
    "vi_nu": "EXAVITQu4vr4xnSDxMaL",  # Sarah (đa ngôn ngữ)
    "vi_nam": "pNInz6obpgDQGcFmaJgB",  # Adam
    "vi_tre": "21m00Tcm4TlvDq8ikWAM",  # Rachel
}


async def tts_elevenlabs(text: str, voice_id: str,
                            output_path: str) -> Dict:
    """ElevenLabs TTS"""
    if not ELEVENLABS_KEY:
        return {"error": "Chưa cấu hình ELEVENLABS_API_KEY"}

    async with httpx.AsyncClient(timeout=60) as client:
        r = await client.post(
            f"https://api.elevenlabs.io/v1/text-to-speech/{voice_id}",
            headers={
                "xi-api-key": ELEVENLABS_KEY,
                "Content-Type": "application/json",
            },
            json={
                "text": text,
                "model_id": "eleven_multilingual_v2",
                "voice_settings": {
                    "stability": 0.5,
                    "similarity_boost": 0.75,
                    "style": 0.3,
                    "use_speaker_boost": True,
                },
            },
        )
        if r.status_code != 200:
            return {"error": r.text}

        with open(output_path, "wb") as f:
            f.write(r.content)

    return {"success": True, "path": output_path, "provider": "elevenlabs"}


# ============ GOOGLE CLOUD TTS ============
GOOGLE_VOICES = {
    "vi_nu": "vi-VN-Standard-A",
    "vi_nam": "vi-VN-Standard-B",
    "vi_nu2": "vi-VN-Wavenet-A",
}


async def tts_google(text: str, voice_name: str,
                        output_path: str) -> Dict:
    """Google Cloud TTS"""
    if not GOOGLE_TTS_KEY:
        return {"error": "Chưa cấu hình GOOGLE_TTS_KEY"}

    async with httpx.AsyncClient(timeout=60) as client:
        r = await client.post(
            f"https://texttospeech.googleapis.com/v1/text:synthesize?key={GOOGLE_TTS_KEY}",
            json={
                "input": {"text": text},
                "voice": {"languageCode": "vi-VN", "name": voice_name},
                "audioConfig": {
                    "audioEncoding": "MP3",
                    "speakingRate": 1.0,
                    "pitch": 0.0,
                },
            },
        )
        if r.status_code != 200:
            return {"error": r.text}

        import base64
        audio_content = base64.b64decode(r.json()["audioContent"])
        with open(output_path, "wb") as f:
            f.write(audio_content)

    return {"success": True, "path": output_path, "provider": "google"}


# ============ COQUI (offline) ============
async def tts_coqui(text: str, output_path: str) -> Dict:
    """
    Coqui TTS — chạy local, miễn phí, cần cài: pip install TTS.
    Hỗ trợ tiếng Việt qua model VietTTS.
    """
    try:
        from TTS.api import TTS
    except ImportError:
        return {"error": "Cần cài: pip install TTS"}

    try:
        # Model tiếng Việt
        tts = TTS(model_name="tts_models/vi/mai/glow-tts", progress_bar=False)
        tts.tts_to_file(text=text, file_path=output_path)
        return {"success": True, "path": output_path, "provider": "coqui"}
    except Exception as e:
        return {"error": str(e)}


# ============ ORCHESTRATOR ============
async def tao_giong_doc(text: str, voice: str = "vi_nu",
                          provider: str = "auto") -> Dict:
    """Tạo file audio đọc text"""
    ts = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
    output = str(MEDIA_DIR / f"voice_{ts}.mp3")

    # Auto chọn provider
    if provider == "auto":
        if ELEVENLABS_KEY:
            provider = "elevenlabs"
        elif GOOGLE_TTS_KEY:
            provider = "google"
        else:
            provider = "coqui"

    if provider == "elevenlabs":
        voice_id = ELEVENLABS_VOICES.get(voice, ELEVENLABS_VOICES["vi_nu"])
        result = await tts_elevenlabs(text, voice_id, output)
    elif provider == "google":
        voice_name = GOOGLE_VOICES.get(voice, GOOGLE_VOICES["vi_nu"])
        result = await tts_google(text, voice_name, output)
    else:
        result = await tts_coqui(text, output)

    if result.get("success"):
        return {
            "url": f"/media/voice/{os.path.basename(output)}",
            "provider": result["provider"],
            "text_length": len(text),
        }
    return result


def tao_script_doc(profile: Dict, chi_so: Dict,
                     phan_tich: Dict) -> str:
    """Tạo script đọc từ dữ liệu phân tích"""
    lines = [
        f"Xin chào. Đây là dự đoán huyền học cho bạn.",
        f"Ngày sinh {profile['ngay_sinh']}, "
        f"mệnh {profile['menh_nap_am']}, con giáp {profile['con_giap']}.",
        "",
        f"Chỉ số tài vận: {int(chi_so.get('tai_van', 50))} trên 100.",
        f"Chỉ số tình duyên: {int(chi_so.get('tinh_duyen', 50))} trên 100.",
        f"Chỉ số sự nghiệp: {int(chi_so.get('su_nghiep', 50))} trên 100.",
        "",
    ]

    if phan_tich.get("tom_tat"):
        lines.append(f"Tổng quan: {phan_tich['tom_tat']}")
        lines.append("")

    actions = phan_tich.get("hanh_dong_cu_the", [])
    if actions:
        lines.append("Hành động cụ thể tuần này:")
        for i, a in enumerate(actions[:3], 1):
            lines.append(f"{i}. {a.get('hanh_dong', '')}")
        lines.append("")

    if phan_tich.get("cau_chuc"):
        lines.append(phan_tich["cau_chuc"])

    lines.append("Cảm ơn bạn đã sử dụng Huyền Bảo AI. Chúc bạn may mắn.")

    return " ".join(lines)
```

```python
def init_lms(db: Session):
    """Seed khóa học mặc định"""
    for course_data in DEFAULT_COURSES:
        existing = db.query(Course).filter_by(slug=course_data["slug"]).first()
        if existing:
            continue

        lessons = course_data.pop("lessons", [])
        course = Course(**course_data)
        db.add(course); db.commit(); db.refresh(course)

        for i, lesson_data in enumerate(lessons, 1):
            lesson = Lesson(
                course_id=course.id, order_num=i,
                ten=lesson_data["ten"],
                noi_dung_md=lesson_data["noi_dung_md"],
                duration_min=lesson_data["duration_min"],
            )
            db.add(lesson)
        db.commit()

async def ask_ai_tutor(user_id: int, question: str,
                         course_id: int = None, lesson_id: int = None,
                         db: Session = None) -> Dict:
    """Hỏi AI tutor về bài học"""

    # Lấy context
    context = ""
    if lesson_id and db:
        lesson = db.query(Lesson).filter_by(id=lesson_id).first()
        if lesson:
            context = f"Bài học hiện tại: {lesson.ten}\n\n{lesson.noi_dung_md}"

    # Lấy profile user để cá nhân hóa
    from database import Profile
    prof = db.query(Profile).filter_by(user_id=user_id).first() if db else None
    profile_info = ""
    if prof:
        p = json.loads(prof.profile_json)
        profile_info = (f"\nHọc viên: sinh {p['ngay_sinh']}, "
                         f"mệnh {p['menh_nap_am']}, {p['con_giap']}")

    prompt = f"""Bạn là giảng viên huyền học nhiệt tình, giảng dễ hiểu cho người mới.

{context}
{profile_info}

Câu hỏi của học viên: {question}

Hãy trả lời:
1. Ngắn gọn, dễ hiểu, có ví dụ cụ thể
2. Nếu liên quan đến mệnh của học viên, đưa ví dụ cá nhân hóa
3. Cuối cùng đề xuất 1-2 bài học tiếp theo nên học

Trả về JSON:
{{
  "tra_loi": "câu trả lời chính",
  "vi_du": ["ví dụ 1", "ví dụ 2"],
  "bai_hoc_tiep_theo": ["tên bài", "tên bài"],
  "cau_hoi_goi_y": ["câu hỏi tiếp theo học viên có thể hỏi"]
}}"""

    from ollama_client import ollama
    raw = await ollama.generate(prompt, temperature=0.7, json_mode=True)
    try:
        answer = json.loads(raw)
    except Exception:
        answer = {"tra_loi": raw, "vi_du": [], "bai_hoc_tiep_theo": []}

    # Lưu chat
    if db:
        chat = AI TutorChat(
            user_id=user_id, course_id=course_id,
            lesson_id=lesson_id, question=question,
            answer=json.dumps(answer, ensure_ascii=False),
        )

    return answer


def danh_gia_tien_do(user_id: int, course_id: int, db: Session) -> Dict:
    """Đánh giá tiến độ học tập"""
    course = db.query(Course).filter_by(id=course_id).first()
    if not course:
        return {"error": "Không tìm thấy khóa học"}

    lessons = db.query(Lesson).filter_by(course_id=course_id).order_by(
        Lesson.order_num
    ).all()

    completed = 0
    total_score = 0
    score_count = 0
    for lesson in lessons:
        prog = db.query(LessonProgress).filter_by(
            user_id=user_id, lesson_id=lesson.id
        ).first()
        if prog and prog.completed:
            completed += 1
            if prog.quiz_score is not None:
                total_score += prog.quiz_score
                score_count += 1

    avg_score = total_score / score_count if score_count else None

    return {
        "course": course.ten,
        "lessons_total": len(lessons),
        "lessons_completed": completed,
        "progress_pct": round(completed / len(lessons) * 100, 1) if lessons else 0,
        "avg_quiz_score": round(avg_score, 1) if avg_score else None,
        "next_lesson": next(
            (l.ten for l in lessons
             if not (db.query(LessonProgress)
                       .filter_by(user_id=user_id, lesson_id=l.id,
                                    completed=True).first())),
            None
        ),
    }
```

```python
                LessonProgress.completed == True,
            ).count()
            enroll.progress_pct = round(completed / total * 100, 1) if total else 0
            enroll.last_activity = datetime.utcnow()
            if enroll.progress_pct >= 100:
                enroll.completed_at = datetime.utcnow()
            db.commit()

    return {"ok": True}


class TutorQuestion(BaseModel):
    question: str
    course_id: int = None
    lesson_id: int = None


@app.post("/api/tutor/ask")
async def api_ask_tutor(data: TutorQuestion,
                          user: User = Depends(get_current_user),
                          db: Session = Depends(get_db)):
    return await ask_ai_tutor(user.id, data.question,
                                data.course_id, data.lesson_id, db)


@app.get("/api/my/learning")
def api_my_learning(user: User = Depends(get_current_user),
                      db: Session = Depends(get_db)):
    enrolls = db.query(Enrollment).filter_by(user_id=user.id).all()
    return {"courses": [{
        "course_id": e.course_id,
        "progress_pct": e.progress_pct,
        "enrolled_at": e.enrolled_at.isoformat(),
        "completed": e.completed_at is not None,
    } for e in enrollments]}
```

| # | Module | File chính | Endpoint chính |
|---|--------|-----------|----------------|
| 1 | 🌐 Cloudflare | `worker.js` | `huyenbao.com/*` |

| 3 | 🤖 Auto-posting | `auto_posting.py` | `/api/social/*` |
| 4 | 💰 Stripe | `monetization.py` | `/api/payments/*` |
| 5 | 🌐 i18n | `i18n.py` | `/api/i18n/*` |
| 6 | 📱 Mobile | `mobile/` | React Native app |
| 7 | 🔗 NFT | `blockchain_nft.py` | `/api/nft/*` |
| 8 | 🎙️ Voice | `voice_ai.py` | `/api/voice/*` |
| 9 | 🏫 LMS | `lms.py` | `/api/courses/*` |

### Điểm đột phá mở rộng lần này

1. **Tên + Giờ sinh**: Thần số học tên (Ngũ cách) + Can Chi giờ → phân tích sâu hơn nhiều lần.

2. **Cloudflare Worker**: Xử lý rate limit, CORS, cache, proxy ngay tại edge → backend nhẹ, latency thấp.

3. **White-label**: 1 backend phục vụ N doanh nghiệp với branding riêng — mô hình SaaS B2B2C.

4. **Auto-posting**: Từ video render → đăng multi-platform tự động → viral marketing engine.

5. **Stripe**: Monetize 4 tier (Free / Premium / VIP / Enterprise) với quota động.

6. **i18n**: Mở rộng ra thị trường quốc tế (Anh, Trung, Nhật) — đặc biệt Trung Quốc cực lớn.

7. **Mobile**: React Native wrap web + native notifications → retention cao hơn.

8. **NFT**: Lưu lá số on-chain → immutable, có thể trade, marketing "web3 huyền học".

9. **Voice AI**: Đọc dự đoán → accessibility + podcast marketing + Telegram voice messages.

10. **LMS + AI Tutor**: Từ app "xem bói" → nền tảng học tập có AI giáo viên → retention dài hạn + doanh thu khóa học.
