```python
def tinh_cung_phi_gioi_tinh(day: int, month: int, year: int, gioi_tinh: str) -> int:
    """
    Tính Cung Phi (Bát Trạch) theo năm sinh + giới tính.
    Nam và nữ có công thức khác nhau.
    """
    # Rút gọn năm sinh
    y = year
    while y > 9:
        y = sum(int(d) for d in str(y))

    if gioi_tinh == "nam":
        cung = 11 - y
        if cung > 9:
            cung -= 9
        # Nếu năm sinh sau 2000, cộng thêm 1
        if year >= 2000:
            cung = cung - 1 if cung > 1 else 9
    else:  # nữ
        cung = y + 4
        if cung > 9:
            cung -= 9
        if year >= 2000:
            cung = cung + 1 if cung < 9 else 1

    # Trung cung (5) → Nam về Khôn (2), Nữ về Cấn (8)
    if cung == 5:
        cung = 2 if gioi_tinh == "nam" else 8

    return cung

def hop_huong_nha(cung_phi: int, huong_nha: str) -> Dict:

def tinh_nien_phi_tinh(nam: int) -> int:
    """Sao nhập trung cung của năm (Niên Tinh)"""
    # Công thức: (11 - (năm rút gọn)) mod 9, cộng 1
    y = nam
    while y > 9:
        y = sum(int(d) for d in str(y))
    sao = 11 - y
    if sao > 9: sao -= 9
    if sao < 1: sao += 9
    return sao


def tinh_nguyet_phi_tinh(nam: int, thang: int) -> int:
    """Sao nhập trung cung của tháng"""
    nien = tinh_nien_phi_tinh(nam)
    # Bảng: tháng 1 âm = sao tháng theo năm
    # Đơn giản hóa: dùng công thức xấp xỉ
    return ((nien + thang - 1) % 9) or 9


def tinh_nhat_phi_tinh(nam: int, thang: int, ngay: int) -> int:
    """Sao nhập trung cung của ngày"""
    nguyet = tinh_nguyet_phi_tinh(nam, thang)
    return ((nguyet + ngay - 1) % 9) or 9


def phi_tinh_bay_cung(sao_trung_cung: int) -> Dict:
    """
    Lường Thiên Xích: bay 9 sao từ trung cung ra 8 hướng.
    """
    # Thứ tự bay: Trung cung → Tây Bắc → Tây → Đông Bắc → Nam →
    # Bắc → Tây Nam → Đông → Đông Nam
    duong_bay = [
        "Trung cung", "Tây Bắc", "Tây", "Đông Bắc",
        "Nam", "Bắc", "Tây Nam", "Đông", "Đông Nam",
    ]
    ket_qua = {}
    for i, huong in enumerate(duong_bay):
        sao = ((sao_trung_cung - 1 + i) % 9) + 1
        ket_qua[huong] = {
            "sao": sao,
            "ten": CUU_TINH[sao]["ten"],
            "hanh": CUU_TINH[sao]["hanh"],
            "cat_hung": CUU_TINH[sao]["cat_hung"],
            "y_nghia": CUU_TINH[sao]["y_nghia"],
        }
    return ket_qua


def phi_tinh_ngay(d: date) -> Dict:
    """Phi tinh ngày hiện tại"""
    am_year = d.year
    sao_ngay = tinh_nhat_phi_tinh(am_year, d.month, d.day)
    return {
        "ngay": d.isoformat(),
        "sao_trung_cung": sao_ngay,
        "sao_info": CUU_TINH[sao_ngay],
        "bay_cung": phi_tinh_bay_cung(sao_ngay),
    }

def danh_gia_dai_van(huong_nha: str, sao_van: int) -> Dict:
    """
    Đại Vận: đánh giá cửa/hướng theo vận 8 (2024-2043), vận 9 (2024+).
    """
    # Vận 9 (2024-2043) — Hỏa vượng
    huong_vuong_van9 = ["Nam", "Đông Nam", "Đông"]  # Hỏa + Mộc sinh Hỏa
    huong_suy_van9 = ["Tây Bắc", "Tây"]  # Kim bị Hỏa khắc

    if huong_nha in huong_vuong_van9:
        return {"danh_gia": "vượng", "diem": 85,
                "ly_do": f"Hướng {huong_nha} được Vận 9 (Hỏa) sinh vượng"}
    elif huong_nha in huong_suy_van9:
        return {"danh_gia": "suy", "diem": 35,
                "ly_do": f"Hướng {huong_nha} bị Vận 9 khắc"}
    return {"danh_gia": "trung bình", "diem": 60,
            "ly_do": f"Hướng {huong_nha} không vượng không suy"}


def danh_gia_toa_huong(huong_toa: str, huong_huong: str,
                        cung_phi: int) -> Dict:
    """
    Tọa Cát / Hướng Cát theo Bát Trạch + Huyền Không.
    """
    hop_toa = hop_huong_nha(cung_phi, huong_toa)
    hop_huong = hop_huong_nha(cung_phi, huong_huong)

    tong_diem = (hop_toa.get("diem", 50) * 0.4 +
                  hop_huong.get("diem", 50) * 0.6)

    return {
        "toa": hop_toa, "huong": hop_huong,
        "diem_tong_hop": round(tong_diem, 1),
        "nhan_xet": "Tọa Cát Hướng Cát — đại cát" if tong_diem >= 75 else
                    "Tọa/Hướng trung bình" if tong_diem >= 50 else
                    "Tọa Hướng xấu — nên hóa giải",
    }


def phan_tich_toan_dien(day: int, month: int, year: int,
                         gioi_tinh: str = "nam",
                         huong_nha: str = "Nam",
                         huong_bep: str = None,
                         huong_phong_ngu: str = None) -> Dict:
    """Phân tích phong thủy toàn diện cho 1 người + nhà ở"""
    cung_phi = tinh_cung_phi_gioi_tinh(day, month, year, gioi_tinh)
    huong = huong_tot_xau(cung_phi)
    nha = hop_huong_nha(cung_phi, huong_nha)

    # Đại vận
    ket_qua["dai_van"] = danh_gia_dai_van(huong_nha,

    return ket_qua
```

```python
def _seeded_random(seed: str):
    import random
    h = int(hashlib.sha256(seed.encode()).hexdigest(), 16)
    return random.Random(h)


def tinh_chi_so_su_kien(profile: Dict, ngay: date,
                         loai: str, su_kien_id: str) -> Dict:
    """
    Tính chỉ số ảnh hưởng (0-100) của 1 sự kiện tới mệnh chủ.
    Dựa trên:
    - Ngày sinh mệnh chủ vs ngày sự kiện
    - Can Chi của ngày sự kiện
    - Mệnh nạp âm
    - Loại sự kiện (trực tiếp cao hơn gián tiếp)
    """
    from lich_van_nien import ngay_can_chi, CAN_NGU_HANH, quan_he_ngu_hanh
    from astrology import tinh_chi_so_su_kien as tinh_cs_base

    can, chi = ngay_can_chi(ngay)
    ngu_hanh_ngay = CAN_NGU_HANH[can]
    menh = profile["menh_nap_am"]
    quan_he = quan_he_ngu_hanh(menh, ngu_hanh_ngay)

    # Base score
    base = tinh_cs_base(profile)
    linh_vuc_chinh = {
        "sinh_nhat": "tinh_than", "ky_hop_dong": "tai_van",
        "phau_thuat": "suc_khoe", "cuoi_hoi": "tinh_duyen",
        "chuyen_nha": "tai_van", "khoi_nghiep": "su_nghiep",
        "thi_cu": "su_nghiep", "dau_tu": "tai_van",
        "them_thanh_vien": "tai_van", "nguoi_than_mat": "suc_khoe",
        "thang_chuc": "su_nghiep", "mat_viec": "su_nghiep",
        "dich_benh": "suc_khoe", "thien_tai": "suc_khoe",
        "kinh_te": "tai_van", "chinh_tri": "su_nghiep",
        "cong_nghe": "su_nghiep", "thoi_tiet_cuc_doan": "suc_khoe",
    }.get(su_kien_id, "gap_go")

    diem_base = base.get(linh_vuc_chinh, 50)

    # Modifier từ ngày
    modifier = {"tương sinh": 15, "tỷ hòa": 10, "trung tính": 0, "tương khắc": -15}[quan_he]

    # Trực tiếp impact mạnh hơn gián tiếp
    loai_info = next(
        (s for cat in LOAI_SU_KIEN.values() for s in cat["items"]
         if s["id"] == su_kien_id), None
    )
    trong_so = loai_info["trong_so"] if loai_info else 0.5
    muc_do = loai_info["muc_do"] if loai_info else "trực tiếp"

    # Sự kiện gián tiếp → ảnh hưởng giảm nhưng phạm vi rộng
    he_so = 1.0 if muc_do == "trực tiếp" else 0.7

    diem_cuoi = (diem_base + modifier) * he_so
    diem_cuoi = max(0, min(100, diem_cuoi))

    return {
        "su_kien_id": su_kien_id,
        "ten": loai_info["ten"] if loai_info else su_kien_id,
        "ngay": ngay.isoformat(),
        "can_chi_ngay": f"{can} {chi}",
        "quan_he_ngu_hanh": quan_he,
        "linh_vuc_chinh": linh_vuc_chinh,
        "muc_do": muc_do,
        "trong_so": trong_so,
        "diem_anh_huong": round(diem_cuoi, 1),
        "du_bao": "tích cực" if diem_cuoi >= 65 else
                   "trung tính" if diem_cuoi >= 45 else "tiêu cực",
        "loai_anh_huong": "trực tiếp" if muc_do == "trực tiếp"
                            else "gián tiếp (lan tỏa)",
    }


def quet_su_kien_tuan(profile: Dict, start: date = None,
                        su_kien_ca_nhan: List[Dict] = None) -> Dict:
    """
    Quét tất cả sự kiện trong tuần tới:
    - Sự kiện cá nhân user khai báo
    - Sự kiện xã hội (dự đoán từ dữ liệu mẫu)
    - Sự kiện chu kỳ (sinh nhật, lễ tết)
    """
    start = start or date.today()
    end = start + timedelta(days=7)

    tat_ca = []

    # 1. Sự kiện cá nhân user khai báo
    if su_kien_ca_nhan:
        for sk in su_kien_ca_nhan:
            ngay_sk = date.fromisoformat(sk["ngay"])
            if start <= ngay_sk <= end:
                ket_qua = tinh_chi_so_su_kien(
                    profile, ngay_sk, "ca_nhan", sk["loai"]
                )
                ket_qua["custom_note"] = sk.get("note", "")
                tat_ca.append(ket_qua)

    # 2. Sinh nhật (nếu trong tuần)
    ngay_sinh = profile["ngay_sinh"]  # "DD/MM/YYYY"
    d, m, _ = map(int, ngay_sinh.split("/"))
    for i in range(7):
        check = start + timedelta(days=i)
        if check.day == d and check.month == m:
            ket_qua = tinh_chi_so_su_kien(profile, check, "ca_nhan", "sinh_nhat")
            tat_ca.append(ket_qua)

    # 3. Sự kiện xã hội mẫu (trong thực tế sẽ call API)
    # Đây là placeholder — production sẽ fetch từ news API
    su_kien_xa_hoi_mau = _lay_su_kien_xa_hoi_mau(start, end)
    for sk in su_kien_xa_hoi_mau:
        ket_qua = tinh_chi_so_su_kien(profile, date.fromisoformat(sk["ngay"]),
                                        "xa_hoi", sk["id"])
        ket_qua["mo_ta"] = sk["mo_ta"]
        tat_ca.append(ket_qua)

    # Sắp xếp theo ngày + điểm ảnh hưởng
    tat_ca.sort(key=lambda x: (x["ngay"], -x["diem_anh_huong"]))

    # Tổng hợp
    return {
        "tu_ngay": start.isoformat(),
        "den_ngay": end.isoformat(),
        "tong_su_kien": len(tat_ca),
        "su_kien": tat_ca,
        "diem_tuan": round(sum(x["diem_anh_huong"] for x in tat_ca) /
                            len(tat_ca), 1) if tat_ca else 50,
        "su_kien_trong_diem": [x for x in tat_ca if x["diem_anh_huong"] >= 70],
        "su_kien_can_chu_y": [x for x in tat_ca if x["diem_anh_huong"] <= 35],
    }


def _lay_su_kien_xa_hoi_mau(start: date, end: date) -> List[Dict]:
    """

def du_bao_dai_han(profile: Dict, nam: int = None) -> Dict:
    """
    Dự đoán sự kiện lớn trong năm — chia theo 4 quý.
    """
    nam = nam or date.today().year
    quy = []
    for q in range(1, 5):
        thang_bat_dau = (q - 1) * 3 + 1
        su_kien_quan_trong = []

        # Sự kiện trọng đại của năm
        for sk_id in ["ky_hop_dong", "dau_tu", "chuyen_viec",
                       "thang_chuc", "khoi_nghiep"]:
            # Điểm dự đoán cho tháng giữa quý
            mid_month = thang_bat_dau + 1
            ngay_test = date(nam, mid_month, 15)
            kq = tinh_chi_so_su_kien(profile, ngay_test, "ca_nhan", sk_id)
            if kq["diem_anh_huong"] >= 60:
                su_kien_quan_trong.append(kq)

        quy.append({
            "quy": q,
            "thang": f"{thang_bat_dau}-{thang_bat_dau + 2}",
            "su_kien_nen_lap_ke_hoach": su_kien_quan_trong,
        })

    return {"nam": nam, "quy": quy}
```

```python
"""
Module kết nối phong thủy với môi trường thực tế:
- Thời tiết (temperature, humidity, air quality)
- Vị trí địa lý (longitude/latitude)
- Ảnh hưởng môi trường đến mệnh chủ
"""
import os
import httpx
from datetime import date, datetime
from typing import Dict, Optional


OPENWEATHER_KEY = os.getenv("OPENWEATHER_API_KEY", "")


async def lay_thoi_tiet(lat: float, lon: float) -> Dict:
    """Lấy thời tiết hiện tại từ OpenWeather"""
    if not OPENWEATHER_KEY:
        return {"error": "Chưa cấu hình OPENWEATHER_API_KEY"}
    url = "https://api.openweathermap.org/data/2.5/weather"
    async with httpx.AsyncClient(timeout=10) as client:
        r = await client.get(url, params={
            "lat": lat, "lon": lon,
            "appid": OPENWEATHER_KEY, "units": "metric", "lang": "vi",
        })
        if r.status_code != 200:
            return {"error": r.text}
        d = r.json()
        return {
            "nhiet_do": d["main"]["temp"],
            "do_am": d["main"]["humidity"],
            "mo_ta": d["weather"][0]["description"],
            "icon": d["weather"][0]["icon"],
            "gio": d["wind"]["speed"],
            "may": d["clouds"]["all"],
            "thanh_pho": d.get("name", ""),
        }


async def lay_chat_luong_khong_khi(lat: float, lon: float) -> Dict:
    """AQI từ OpenWeather"""
    if not OPENWEATHER_KEY:
        return {"error": "Chưa cấu hình"}
    url = "https://api.openweathermap.org/data/2.5/air_pollution"
    async with httpx.AsyncClient(timeout=10) as client:
        r = await client.get(url, params={
            "lat": lat, "lon": lon, "appid": OPENWEATHER_KEY,
        })
        if r.status_code != 200:
            return {"error": r.text}
        d = r.json()["list"][0]
        aqi = d["main"]["aqi"]  # 1-5
        return {
            "aqi": aqi,
            "muc_do": ["Tốt", "Khá", "Trung bình", "Kém", "Nguy hại"][aqi-1],
            "pm2_5": d["components"]["pm2_5"],
            "pm10": d["components"]["pm10"],
            "o3": d["components"]["o3"],
            "no2": d["components"]["no2"],
            "so2": d["components"]["so2"],
            "co": d["components"]["co"],
        }


# ============ ẢNH HƯỞNG MÔI TRƯỜNG → MỆNH CHỦ ============
def danh_gia_anh_huong_moi_truong(profile: Dict,
                                    thoi_tiet: Dict,
                                    aqi: Dict) -> Dict:
    """
    Đánh giá môi trường hiện tại ảnh hưởng thế nào đến mệnh chủ.
    """
    menh = profile["menh_nap_am"]
    ket_qua = {"diem_tong_hop": 50, "nhan_xet": [], "hanh_dong": []}

    # 1. Ngũ hành môi trường vs mệnh
    # Nhiệt độ cao → Hỏa vượng; thấp → Thủy vượng
    if "nhiet_do" in thoi_tiet:
        t = thoi_tiet["nhiet_do"]
        if t > 35:
            hanh_mt = "Hỏa"
            ket_qua["nhan_xet"].append(f"Nhiệt độ cao {t}°C — Hỏa vượng")
        elif t < 15:
            hanh_mt = "Thủy"
            ket_qua["nhan_xet"].append(f"Nhiệt độ thấp {t}°C — Thủy vượng")
        else:
            hanh_mt = None
            ket_qua["nhan_xet"].append(f"Nhiệt độ dễ chịu {t}°C")

        if hanh_mt:
            from lich_van_nien import quan_he_ngu_hanh
            qh = quan_he_ngu_hanh(menh, hanh_mt)
            if qh == "tương sinh":
                ket_qua["diem_tong_hop"] += 10
                ket_qua["nhan_xet"].append(f"{menh} tương sinh {hanh_mt} — thuận")
            elif qh == "tương khắc":
                ket_qua["diem_tong_hop"] -= 10
                ket_qua["nhan_xet"].append(f"{menh} khắc {hanh_mt} — cần chú ý")
                ket_qua["hanh_dong"].append(
                    f"Hạn chế ra ngoài giờ cao điểm, uống nhiều nước"
                )

    # 2. Độ ẩm
    if "do_am" in thoi_tiet:
        h = thoi_tiet["do_am"]
        if h > 85:
            ket_qua["diem_tong_hop"] -= 5
            ket_qua["nhan_xet"].append(f"Độ ẩm cao {h}% — Thủy khí nặng")
            ket_qua["hanh_dong"].append("Dùng máy hút ẩm, tránh ẩm mốc")
        elif h < 30:
            ket_qua["diem_tong_hop"] -= 5
            ket_qua["nhan_xet"].append(f"Không khí khô {h}% — Kim/Hỏa khô")
            ket_qua["hanh_dong"].append("Uống đủ nước, dùng máy tạo ẩm")

    # 3. Chất lượng không khí
    if "aqi" in aqi:
        aqi_val = aqi["aqi"]
        if aqi_val >= 4:
            ket_qua["diem_tong_hop"] -= 20
            ket_qua["nhan_xet"].append(
                f"Không khí {aqi['muc_do']} (PM2.5: {aqi['pm2_5']}) — nguy hiểm"
            )
            ket_qua["hanh_dong"].extend([
                "Đeo khẩu trang N95 khi ra ngoài",
                "Đóng cửa sổ, dùng máy lọc không khí",
                "Hạn chế tập thể dục ngoài trời",
            ])
        elif aqi_val == 3:
            ket_qua["diem_tong_hop"] -= 10
            ket_qua["nhan_xet"].append(f"Không khí {aqi['muc_do']} — nhạy cảm")
            ket_qua["hanh_dong"].append("Người có bệnh hô hấp nên ở trong nhà")

    # 4. Mưa / gió
    if thoi_tiet.get("mo_ta"):
        mt = thoi_tiet["mo_ta"].lower()
        if "mưa" in mt or "rain" in mt:
            ket_qua["nhan_xet"].append("Có mưa — Thủy khí vượng")
            ket_qua["hanh_dong"].append("Mang áo mưa, tránh đi xa giờ mưa lớn")
        if thoi_tiet.get("gio", 0) > 10:
            ket_qua["nhan_xet"].append(f"Gió mạnh {thoi_tiet['gio']} m/s — Phong khí")
            ket_qua["hanh_dong"].append("Tránh đứng dưới cây lớn, biển hiệu")

    ket_qua["diem_tong_hop"] = max(0, min(100, ket_qua["diem_tong_hop"]))
    return ket_qua


# ============ VỊ TRÍ PHONG THỦY ============
def goi_y_vi_tri_tot(profile: Dict, huong_tot: List[str]) -> Dict:

        "Hỏa": ["Nam"],

    }
    return {
```

## 🔗 MODULE 4: Kết Nối Mệnh Chủ ↔ Sự Kiện — `ket_noi.py`

```python
"""
Module kết nối tất cả: mệnh chủ + phong thủy + sự kiện + môi trường
→ Bức tranh tổng thể + dự đoán cascade (domino effect)
"""
from datetime import date, timedelta
from typing import Dict, List
import json


def buc_tranh_tong_the(profile: Dict, phong_thuy: Dict,
                         su_kien_tuan: Dict,
                         moi_truong: Dict = None) -> Dict:
    """
    Tạo bức tranh tổng thể: mọi yếu tố ảnh hưởng đến mệnh chủ trong tuần.
    """
    # 1. Điểm nền từ mệnh
    from astrology import tinh_chi_so_su_kien as cs_base
    diem_nen = cs_base(profile)

    # 2. Phong thủy modifier
    phong_thuy_modifier = 0
    if phong_thuy.get("huong_nha", {}).get("tot"):
        phong_thuy_modifier += 10
    else:
        phong_thuy_modifier -= 10

    # 3. Sự kiện modifier
    su_kien_modifier = 0
    if su_kien_tuan.get("su_kien_trong_diem"):
        su_kien_modifier += len(su_kien_tuan["su_kien_trong_diem"]) * 5
    if su_kien_tuan.get("su_kien_can_chu_y"):
        su_kien_modifier -= len(su_kien_tuan["su_kien_can_chu_y"]) * 5

    # 4. Môi trường modifier
    moi_truong_modifier = 0
    if moi_truong:
        mt_score = moi_truong.get("anh_huong", {}).get("diem_tong_hop", 50)
        moi_truong_modifier = (mt_score - 50) * 0.3

    # 5. Tổng hợp
    diem_cuoi = {}
    for linh_vuc, base_score in diem_nen.items():
        final = (base_score +
                  phong_thuy_modifier * 0.3 +
                  su_kien_modifier * 0.4 +
                  moi_truong_modifier * 0.3)
        diem_cuoi[linh_vuc] = round(max(0, min(100, final)), 1)

    return {
        "ngay": date.today().isoformat(),
        "diem_tong_hop": diem_cuoi,
        "diem_trung_binh": round(sum(diem_cuoi.values()) / len(diem_cuoi), 1),
        "yeu_to_anh_huong": {
            "phong_thuy_modifier": phong_thuy_modifier,
            "su_kien_modifier": su_kien_modifier,
            "moi_truong_modifier": round(moi_truong_modifier, 1),
        },
        "canh_bao_cascade": phat_hien_cascade(diem_cuoi, su_kien_tuan),
    }


def phat_hien_cascade(diem: Dict, su_kien_tuan: Dict) -> List[Dict]:
    """
    Phát hiện domino effect: 1 sự kiện xấu có thể kéo theo nhiều hệ quả.
    """
    canh_bao = []

    # Nếu sức khỏe yếu → ảnh hưởng tài vận, sự nghiệp
    if diem.get("suc_khoe", 50) < 40:
        canh_bao.append({
            "loai": "cascade_suc_khoe",
            "mo_ta": "Sức khỏe yếu có thể kéo theo sự nghiệp và tài vận suy giảm",
            "doi_tuong_anh_huong": ["su_nghiep", "tai_van"],
            "hanh_dong": [
                "Ưu tiên nghỉ ngơi, khám sức khỏe",
                "Trì hoãn quyết định lớn về tài chính",
            ],
        })

    # Tài vận yếu + sự nghiệp yếu → nguy cơ mất việc
    if diem.get("tai_van", 50) < 40 and diem.get("su_nghiep", 50) < 40:
        canh_bao.append({
            "loai": "cascade_tai_chinh_su_nghiep",
            "mo_ta": "Tài chính + sự nghiệp đồng thời yếu — nguy cơ khủng hoảng",
            "doi_tuong_anh_huong": ["tinh_than", "gia_dinh"],
            "hanh_dong": [
                "Cắt giảm chi tiêu không cần thiết",
                "Không vay nợ mới",
                "Tìm nguồn thu nhập dự phòng",
            ],
        })

    # Tình duyên + tinh thần yếu → nguy cơ trầm cảm
    if diem.get("tinh_duyen", 50) < 40 and diem.get("tinh_than", 50) < 40:
        canh_bao.append({
            "loai": "cascade_tinh_cam",
            "mo_ta": "Tình cảm + tinh thần yếu — cần chăm sóc bản thân",
            "doi_tuong_anh_huong": ["suc_khoe", "cong_viec"],
            "hanh_dong": [
                "Chia sẻ với người tin tưởng",
                "Thiền/tập thể dục đều đặn",
                "Tránh quyết định lớn về tình cảm lúc này",
            ],
        })

    # Sự kiện lớn sắp tới + điểm nền thấp
    for sk in su_kien_tuan.get("su_kien", []):
        if sk["diem_anh_huong"] < 35 and sk["trong_so"] >= 0.8:
            canh_bao.append({
                "loai": "su_kien_quan_trong_diem_thap",
                "mo_ta": f"Sự kiện '{sk['ten']}' ngày {sk['ngay']} có điểm thấp",
                "doi_tuong_anh_huong": [sk["linh_vuc_chinh"]],
                "hanh_dong": [
                    f"Chuẩn bị kỹ cho sự kiện, có phương án B",
                    "Tránh quyết định quan trọng đúng ngày này",
                ],
            })

    return canh_bao


def tao_lich_hanh_dong(profile: Dict, su_kien_tuan: Dict,

        else:
            hanh_dong["nen_lam"] = [
                "Trì hoãn nếu có thể",

                "Nghỉ ngơi, tĩnh tâm",
            ]
            hanh_dong["tranh"] = [
                "Ký kết hợp đồng lớn",
                "Đầu tư rủi ro",
                "Tranh cãi",
                "Đi xa nếu không cần",
            ]

        lich.append(hanh_dong)

    return lich
```

## 🌟 Điểm đột phá của mở rộng này

1. **Từ cá nhân → hệ sinh thái**: Mệnh chủ không đứng một mình — chịu ảnh hưởng từ nhà, khu vực, xã hội, vũ trụ.

2. **Cascade effect (domino)**: Phát hiện chuỗi hệ quả — sức khỏe yếu → sự nghiệp trì trệ → tài vận sa sút. Cảnh báo sớm.

3. **Trực tiếp vs Gián tiếp**: Phân biệt rõ — sự kiện cá nhân (điểm cao) vs sự kiện xã hội (điểm thấp nhưng lan tỏa rộng).

4. **Phong thủy động**: Kết hợp Bát trạch tĩnh (cung phi) + Huyền không động (phi tinh thay đổi theo ngày/tháng/năm).

5. **Môi trường realtime**: Thời tiết + AQI ảnh hưởng ngũ hành mệnh chủ → khuyến nghị cụ thể từng ngày.

6. **LLM tổng hợp**: AI không chỉ luận từng phần mà kết nối mọi yếu tố → nhận định holistic + lịch hành động 7 ngày.
